import hashlib,json,os,re,stat,sys

def need(ok,why):
 if not ok:raise RuntimeError(why)
def digest(value):return hashlib.sha256(json.dumps(value,sort_keys=True,separators=(',',':')).encode()).hexdigest()
def args_hash(argv):return hashlib.sha256(b''.join(v.encode()+b'\0' for v in argv)).hexdigest()
def descriptors():
 out=[]
 for name in os.listdir('/proc/self/fd'):
  if int(name)<3:continue
  try:os.fstat(int(name))
  except OSError as error:
   if error.errno==9:continue
   raise
  out.append(int(name))
 return sorted(out)
def metadata(s):return {'device':s.st_dev,'inode':s.st_ino,'mode':stat.S_IMODE(s.st_mode),'uid':s.st_uid,'gid':s.st_gid,'kind':'directory' if stat.S_ISDIR(s.st_mode) else 'symlink' if stat.S_ISLNK(s.st_mode) else 'other'}
def decode(s):return re.sub(r'\\([0-7]{3})',lambda m:chr(int(m.group(1),8)),s)

need(len(sys.argv)>=10,'Bootstrap command is incomplete')
outer=json.loads(sys.argv[1]);code_sha=sys.argv[2];tail_sha=sys.argv[3];tail=sys.argv[4:]
need(set(outer)=={'mnt','pid','net'} and all(re.fullmatch(k+r':\[\d+\]',v) for k,v in outer.items()),'Outer namespace identities differ')
need(sys.orig_argv[1:3]==['-B','-c'] and hashlib.sha256(sys.orig_argv[3].encode()).hexdigest()==code_sha,'Bootstrap bytes differ')
need(args_hash(tail)==tail_sha and tail[:3]==['/usr/bin/taskset','-c','24'] and tail[3].startswith('/') and tail[4]=='-B','Original command tail differs')
need(descriptors()==[],'Bootstrap inherited a descriptor')
current={k:os.readlink('/proc/self/ns/'+k) for k in outer}
need(all(current[k]!=outer[k] for k in outer),'Bootstrap namespace is not fresh')
need(os.readlink('/proc/1/ns/pid')==current['pid'],'Bootstrap proc belongs to another PID namespace')
rows=[]
with open('/proc/self/mountinfo') as stream:
 for line in stream:
  a,sep,b=line.partition(' - ');left,right=a.split(),b.split();need(sep and len(left)>=6 and len(right)>=3,'Malformed bootstrap mount table')
  rows.append({'id':int(left[0]),'parent':int(left[1]),'target':decode(left[4]),'options':left[5].split(','),'filesystem':right[0]})
need(len({r['id'] for r in rows})==len(rows),'Duplicate bootstrap mount identity')
roots=[r for r in rows if r['target']=='/'];devs=[r for r in rows if r['target']=='/dev'];procs=[r for r in rows if r['target']=='/proc']
need(len(roots)==len(devs)==len(procs)==1 and roots[0]['filesystem']=='tmpfs' and devs[0]['filesystem']=='tmpfs' and procs[0]['filesystem']=='proc','Expected private root/dev/proc mounts required')
need('ro' in roots[0]['options'] and 'rw' in devs[0]['options'] and bool(os.statvfs('/').f_flag&os.ST_RDONLY) and not bool(os.statvfs('/dev').f_flag&os.ST_RDONLY),'Private root/dev access mode differs')
need(not any(r['target']==p or r['target'].startswith(p+'/') for r in rows for p in ('/dev/shm','/dev/core')),'Unexpected shm/core mount')
environment_sha=digest(dict(os.environ));fd=os.open('/dev',os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC)
try:
 dev=os.fstat(fd);need(stat.S_ISDIR(dev.st_mode) and dev.st_dev!=os.stat('/').st_dev and dev.st_uid==os.geteuid(),'Private dev identity differs')
 need('console' not in os.listdir(fd),'Unexpected console device')
 shm=os.stat('shm',dir_fd=fd,follow_symlinks=False);core=os.stat('core',dir_fd=fd,follow_symlinks=False)
 need(stat.S_ISDIR(shm.st_mode) and stat.S_IMODE(shm.st_mode)==0o755 and shm.st_uid==os.geteuid() and shm.st_dev==dev.st_dev,'Unexpected private shm identity')
 need(stat.S_ISLNK(core.st_mode) and core.st_uid==os.geteuid() and core.st_dev==dev.st_dev and os.readlink('core',dir_fd=fd)=='/proc/kcore','Unexpected private core identity')
 shm_fd=os.open('shm',os.O_RDONLY|os.O_DIRECTORY|os.O_NOFOLLOW|os.O_CLOEXEC,dir_fd=fd)
 try:
  need((os.fstat(shm_fd).st_dev,os.fstat(shm_fd).st_ino)==(shm.st_dev,shm.st_ino) and os.listdir(shm_fd)==[],'Private shm is nonempty or replaced')
 finally:os.close(shm_fd)
 need(metadata(os.stat('shm',dir_fd=fd,follow_symlinks=False))==metadata(shm) and metadata(os.stat('core',dir_fd=fd,follow_symlinks=False))==metadata(core) and os.readlink('core',dir_fd=fd)=='/proc/kcore','Private defaults changed before removal')
 os.rmdir('shm',dir_fd=fd);os.unlink('core',dir_fd=fd)
 need(not {'shm','core'}&set(os.listdir(fd)),'Private default removal did not complete')
finally:os.close(fd)
need(descriptors()==[] and digest(dict(os.environ))==environment_sha,'Bootstrap altered descriptors or environment')
print(json.dumps({'schema':'orbit-poc-private-dev-bootstrap-result-v1','status':'passed','bootstrap_sha256':code_sha,'original_command_sha256':tail_sha,'command':tail,'environment_sha256':environment_sha,'outer_namespaces':outer,'namespaces':current,'mounts':{'root':roots[0],'dev':devs[0],'proc':procs[0]},'dev':metadata(dev),'removed':[{'path':'/dev/shm',**metadata(shm)},{'path':'/dev/core','target':'/proc/kcore',**metadata(core)}],'descriptors':[],'original_command_preserved':True}),flush=True)
os.execve(tail[0],tail,dict(os.environ))
